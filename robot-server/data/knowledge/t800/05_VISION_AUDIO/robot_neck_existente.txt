#!/usr/bin/env python3
"""Publish a 3-DoF IO teleoperation head/torso test command.

该脚本用于测试 `teleoperation_io.cpp` 中的 `target_joint_from_vr_sub_`。

发布接口约定：
- topic: `/io_teleop/target_joint_from_vr`
- message type: `sensor_msgs/msg/JointState`
- position 维度: 3
- 顺序: `[head_pitch_src, head_yaw, torso]`

测试信号：
- 第 0 维 `head_pitch_src` 发布余弦波，上限 0.45 rad，下限 0 rad
- 第 1、2 维保持以 0 rad 为中心、振幅 0.5 rad 的正弦波
- 所有通道频率均为 0.5 Hz
- 发布频率为 100 Hz
- 正弦运动执行 3 个完整周期后自动停止节点
"""

import math

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState


TARGET_TOPIC = "/io_teleop/target_joint_from_vr"
PUBLISH_HZ = 100.0
SINE_FREQ_HZ = 0.5
SINE_AMPLITUDE_RAD = 0.5
TOTAL_SINE_CYCLES = 3
HEAD_PITCH_SRC_MAX_RAD = 0.45
HEAD_PITCH_SRC_MIN_RAD = 0.0
HEAD_PITCH_SRC_CENTER_RAD = (HEAD_PITCH_SRC_MAX_RAD + HEAD_PITCH_SRC_MIN_RAD) / 2.0
HEAD_PITCH_SRC_AMPLITUDE_RAD = (HEAD_PITCH_SRC_MAX_RAD - HEAD_PITCH_SRC_MIN_RAD) / 2.0

JOINT_NAMES = ["head_pitch_src", "head_yaw", "torso"]


class HeadTorsoPublisher(Node):
    """Publish sinusoidal head and torso targets for IO teleoperation testing."""

    def __init__(self) -> None:
        super().__init__("io_teleop_head_torso_pub")

        self.publisher_ = self.create_publisher(JointState, TARGET_TOPIC, 10)
        self.timer_ = self.create_timer(1.0 / PUBLISH_HZ, self.timer_callback)
        self.start_time_sec = self.get_clock().now().nanoseconds / 1e9
        self.total_motion_duration_sec = TOTAL_SINE_CYCLES / SINE_FREQ_HZ
        self.motion_finished = False

        self.get_logger().info(
            f"Publishing {TARGET_TOPIC} at {PUBLISH_HZ:.0f} Hz with {SINE_FREQ_HZ:.1f} Hz, "
            f"head_pitch_src in [{HEAD_PITCH_SRC_MIN_RAD:.3f}, {HEAD_PITCH_SRC_MAX_RAD:.3f}] rad and "
            f"other targets at +/-{SINE_AMPLITUDE_RAD:.1f} rad. "
            f"Will stop automatically after {TOTAL_SINE_CYCLES} sine cycles "
            f"({self.total_motion_duration_sec:.1f} s)."
        )

    def publish_target(self, head_pitch_src: float, head_yaw: float, torso: float) -> None:
        """Publish one `/io_teleop/target_joint_from_vr` JointState message."""
        msg = JointState()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.name = list(JOINT_NAMES)
        msg.position = [head_pitch_src, head_yaw, torso]
        msg.velocity = []
        msg.effort = []

        self.publisher_.publish(msg)

    def timer_callback(self) -> None:
        """Publish a 3-DoF sinusoidal JointState command."""
        if self.motion_finished:
            return

        now_sec = self.get_clock().now().nanoseconds / 1e9
        elapsed_sec = now_sec - self.start_time_sec

        # 运行满 3 个完整周期后，显式发布一帧中心位参考，然后自动退出。
        # 这里显式发布一次，
        # 可以避免由于定时器采样误差导致停在微小残余偏移上。
        if elapsed_sec >= self.total_motion_duration_sec:
            self.motion_finished = True
            self.publish_target(HEAD_PITCH_SRC_CENTER_RAD, 0.0, 0.0)
            self.timer_.cancel()
            self.get_logger().info(
                f"Completed {TOTAL_SINE_CYCLES} sine cycles, returned to center pose, stopping node."
            )
            return

        phase = 2.0 * math.pi * SINE_FREQ_HZ * elapsed_sec
        unit_sine = math.sin(phase)
        # 使用 1-cos 形式，让第 0 维从 0rad 起步，同时保持范围 [0, 0.45]。
        head_pitch_src = HEAD_PITCH_SRC_CENTER_RAD - HEAD_PITCH_SRC_AMPLITUDE_RAD * math.cos(phase)
        sine_value = SINE_AMPLITUDE_RAD * unit_sine
        self.publish_target(head_pitch_src, sine_value, sine_value)


def main(args=None) -> None:
    rclpy.init(args=args)
    node = HeadTorsoPublisher()
    try:
        while rclpy.ok() and not node.motion_finished:
            rclpy.spin_once(node, timeout_sec=0.1)
    finally:
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()


if __name__ == "__main__":
    main()
