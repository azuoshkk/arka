def launch_neck_controller():
    """Inicia automaticamente o neck_controller.py na porta 5055 como subprocesso gerenciado com ambiente ROS 2."""
    global neck_controller_process
    possible_paths = [
        os.path.join(os.path.dirname(os.path.abspath(__file__)), "neck_controller.py"),
        os.path.expanduser("~/scripts-ipq/beniFiles/neck_controller.py"),
        "/home/ubuntu/scripts-ipq/beniFiles/neck_controller.py",
        os.path.abspath("neck_controller.py")
    ]
    
    script_path = None
    for path in possible_paths:
        if os.path.exists(path):
            script_path = os.path.abspath(path)
            break
            
    if script_path:
        try:
            log_file = open('/tmp/neck_controller.log', 'w')
            script_dir = os.path.dirname(script_path) or "."
            
            # Sourcing resiliente do ROS 2 e do workspace engineai do robô com rede CycloneDDS
            env_cmd = (
                f"export ROS_DOMAIN_ID=69; "
                f"export RMW_IMPLEMENTATION=rmw_cyclonedds_cpp; "
                f"export ROS_LOCALHOST_ONLY=0; "
                f"[ -f /opt/ros/humble/setup.bash ] && source /opt/ros/humble/setup.bash 2>/dev/null; "
                f"[ -f /opt/ros/foxy/setup.bash ] && source /opt/ros/foxy/setup.bash 2>/dev/null; "
                f"for ws in ~/engineai_workspace-2 /Users/beniprado/Downloads/engineai_workspace-2 /home/ubuntu/engineai_workspace-2; do "
                f"  if [ -f \"$ws/install/setup.bash\" ]; then "
                f"    source \"$ws/install/setup.bash\" 2>/dev/null; "
                f"    [ -f \"$ws/src/interface_example/scripts/env.sh\" ] && source \"$ws/src/interface_example/scripts/env.sh\" 2>/dev/null; "
                f"    break; "
                f"  fi; "
                f"done; "
                f"exec {sys.executable} \"{script_path}\""
            )

            neck_controller_process = subprocess.Popen(
                ["bash", "-c", env_cmd],
                cwd=script_dir,
                stdout=log_file,
                stderr=subprocess.STDOUT,
                preexec_fn=os.setsid if hasattr(os, 'setsid') else None
            )
            print(f"[NECK] ⚡ Iniciando neck_controller.py na porta 5055 ({script_path})...")
            
            # Aguarda até 3 segundos para validar se a porta 5055 subiu
            import urllib.request
            server_online = False
            for _ in range(15):
                time.sleep(0.2)
                if neck_controller_process.poll() is not None:
                    break
                try:
                    req = urllib.request.Request("http://127.0.0.1:5055/health")
                    with urllib.request.urlopen(req, timeout=0.5) as resp:
                        if resp.status == 200:
                            server_online = True
                            break
                except Exception:
                    pass

            if server_online:
                print(f"[NECK] ✅ neck_controller.py está ONLINE e respondendo na porta 5055!")
            else:
                ret = neck_controller_process.poll()
                print(f"[NECK] ⚠️ neck_controller.py inicializado (código de saída ou em progresso: {ret}).")
                if os.path.exists('/tmp/neck_controller.log'):
                    with open('/tmp/neck_controller.log', 'r') as f:
                        log_content = f.read().strip()
                        if log_content:
                            print(f"[NECK LOG] Detalhes:\n{log_content}")

            return neck_controller_process
        except Exception as e:
            print(f"[NECK] ❌ Falha crítica ao iniciar neck_controller.py: {e}")
    else:
        print("[NECK] ⚠️ Aviso: neck_controller.py não encontrado nas localizações padrão.")
    return None
